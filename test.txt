import os
import numpy as np
import matplotlib.pyplot as plt
from datasets import load_from_disk
import argparse
from PIL import Image  # <-- added import

# Argument parsing to pass dataset path from CLI
parser = argparse.ArgumentParser(description="Analyze spectrogram statistics from an HF dataset.")
parser.add_argument("--dataset_path", type=str, default=".../data/hf",
                    help="Path to the HF dataset directory.")
parser.add_argument("--batch_size", type=int, default=10,
                    help="Batch size for processing spectrograms.")
args = parser.parse_args()

# Load the HF dataset
print("Loading dataset from:", args.dataset_path)
hf_dataset = load_from_disk(args.dataset_path)

# Get dataset info
dataset_size = len(hf_dataset)
print(f"Dataset size: {dataset_size}")

# Initialize variables for computing statistics
running_count = 0
global_min = float('inf')
global_max = float('-inf')

# First pass: find global min/max for histogram bins
print("First pass: determining global min/max...")
for i in range(0, dataset_size, args.batch_size):
    batch_end = min(i + args.batch_size, dataset_size)
    batch_indices = list(range(i, batch_end))
    
    for idx in batch_indices:
        img = hf_dataset[idx]["spectrogram"]
        if isinstance(img, Image.Image):
            img_array = np.array(img)
        else:
            img_array = img
        
        batch_min = np.min(img_array)
        batch_max = np.max(img_array)
        global_min = min(global_min, batch_min)
        global_max = max(global_max, batch_max)

# Print summary after the first pass
print(f"First pass completed. Global Min: {global_min:.2f}, Global Max: {global_max:.2f}")

# Initialize histogram bins
n_bins = 100
hist_bins = np.linspace(global_min, global_max, n_bins + 1)
hist_counts = np.zeros(n_bins)

# Initialize variables for global statistics
total_sum = 0
total_sum_sq = 0
total_pixel_count = 0

# Process data in batches
for i in range(0, dataset_size, args.batch_size):
    batch_end = min(i + args.batch_size, dataset_size)
    batch_indices = list(range(i, batch_end))
    
    # Initialize batch statistics
    batch_pixel_count = 0
    batch_sum = 0
    batch_sum_sq = 0
    batch_min = float('inf')
    batch_max = float('-inf')
    
    for idx in batch_indices:
        img = hf_dataset[idx]["spectrogram"]
        if isinstance(img, Image.Image):
            img_array = np.array(img)
        else:
            img_array = img
        
        # Update batch statistics
        batch_sum += np.sum(img_array)
        batch_sum_sq += np.sum(img_array**2)
        batch_pixel_count += img_array.size
        batch_min = min(batch_min, np.min(img_array))
        batch_max = max(batch_max, np.max(img_array))
        
        # Update global statistics
        total_sum += np.sum(img_array)
        total_sum_sq += np.sum(img_array**2)
        total_pixel_count += img_array.size
        running_count += 1
        
        # Update histogram
        img_hist, _ = np.histogram(img_array.flatten(), bins=hist_bins)
        hist_counts += img_hist
    
    # Compute batch statistics
    batch_mean = batch_sum / batch_pixel_count
    batch_variance = (batch_sum_sq / batch_pixel_count) - (batch_mean**2)
    batch_std = np.sqrt(batch_variance)
    
    # Print batch statistics in one line
    print(f"Processing batch {i//args.batch_size + 1}: samples {i+1}-{batch_end} | "
          f"Mean: {batch_mean:.2f}, Std: {batch_std:.2f}, Min: {batch_min:.2f}, Max: {batch_max:.2f}")

print(f"Total samples processed: {running_count}")
print(f"Total pixels processed: {total_pixel_count}")

# Compute final statistics
global_mean_scalar = total_sum / total_pixel_count
global_variance = (total_sum_sq / total_pixel_count) - (global_mean_scalar**2)
global_std = np.sqrt(global_variance)

print(f"Global Mean: {global_mean_scalar:.2f}, Std: {global_std:.2f}, Min: {global_min:.2f}, Max: {global_max:.2f}")

# Compute quantiles from histogram
cumulative_counts = np.cumsum(hist_counts)
total_count = cumulative_counts[-1]
q25_idx = np.searchsorted(cumulative_counts, 0.25 * total_count)
q50_idx = np.searchsorted(cumulative_counts, 0.50 * total_count)
q75_idx = np.searchsorted(cumulative_counts, 0.75 * total_count)

q25 = hist_bins[q25_idx]
q50 = hist_bins[q50_idx]
q75 = hist_bins[q75_idx]
print(f"25th Quantile: {q25:.2f}, 50th Quantile: {q50:.2f}, 75th Quantile: {q75:.2f}")

# Get first sample for profile analysis
first_sample = np.array(hf_dataset[0]["spectrogram"])
time_profile = np.mean(first_sample, axis=0)
frequency_profile = np.mean(first_sample, axis=1)

# Plot histogram of all amplitude values
plt.figure(figsize=(8,5))
bin_centers = (hist_bins[:-1] + hist_bins[1:]) / 2
plt.bar(bin_centers, hist_counts, width=np.diff(hist_bins), color='skyblue', edgecolor='black', alpha=0.7)
plt.title("Histogram of All Amplitude Values")
plt.xlabel("Amplitude")
plt.ylabel("Frequency")
plt.tight_layout()
plt.show()

# Plot time and frequency profiles for the first spectrogram sample
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10,8))

ax1.plot(time_profile, color='red')
ax1.set_title("Time-wise Mean Amplitude (First Sample)")
ax1.set_xlabel("Time")
ax1.set_ylabel("Mean Amplitude")

ax2.plot(frequency_profile, color='blue')
ax2.set_title("Frequency-wise Mean Amplitude (First Sample)")
ax2.set_xlabel("Frequency")
ax2.set_ylabel("Mean Amplitude")
plt.tight_layout()
plt.show()
